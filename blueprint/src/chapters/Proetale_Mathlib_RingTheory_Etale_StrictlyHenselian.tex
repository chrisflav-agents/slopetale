This chapter (placeholder) records the two key structural results
about étale algebras over a strictly Henselian local ring $A$.

\begin{theorem}
  \label{thm:strictly-henselian-good-retraction}
  \lean{Algebra.Etale.strictlyHenselian_good_retraction}
  \leanok
  Let $A$ be a strictly Henselian local ring, $A \to B$ étale, and
  $\mathfrak{n} \subset B$ a maximal ideal lying over the maximal
  ideal $\mathfrak{m}$ of $A$. Then there exists an $A$-algebra
  retraction $s : B \to A$ with $\mathfrak{n} = s^{-1}(\mathfrak{m})$.
\end{theorem}

\begin{theorem}
  \label{thm:etale-over-strictly-henselian-localization-isom}
  \lean{Algebra.Etale.bijective_localRingHom_of_strictlyHenselian}
  \leanok
  Under the same hypotheses, the canonical local ring homomorphism
  $\mathrm{Loc}_{\mathfrak{m}}(A) \to B_{\mathfrak{n}}$ is bijective.
  Equivalently, the composite $A \to B_{\mathfrak{n}}$ is an
  isomorphism.
\end{theorem}
